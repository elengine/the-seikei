## 型(`src/core/domain/types.ts`)

```ts
export type ColorId = string;
export interface YarnColor { id: ColorId; name: string; hex: string; symbol: string; }

export type YarnTypeId = string;
export interface YarnType { id: YarnTypeId; color: ColorId; hinban: string; spec: string; }

export interface StripeRun { yarn: YarnTypeId; count: number; }
export type StripePlan = StripeRun[];

export interface Pattern {
  id: string; name: string; plan: StripePlan; weft: ColorId;
  era: { from: number; to: number }; description: string; difficulty: 1 | 2 | 3;
}

export interface CreelPuzzle {
  id: string;          // 's1' など
  stage: 1 | 2 | 3 | 4 | 5;
  patternId: string;
  rows: number;        // 段
  cols: number;        // 軸(1〜8)
}
```

## 読み込みと検査(`src/core/content/content.ts`)

```ts
export interface Content {
  colors: Map<ColorId, YarnColor>;
  yarns: Map<YarnTypeId, YarnType>;
  patterns: Map<string, Pattern>;
  creelPuzzles: CreelPuzzle[];    // stage の昇順
  problems: string[];             // 読み飛ばした項目の説明(管理者向け)
}
export function loadContent(raw: { colors: unknown; yarns: unknown; patterns: unknown; creelPuzzles: unknown }): Content;
export function getContent(): Content; // 4つの JSON を import して loadContent した結果を返す(1回だけ計算して覚える)
```

- 形が正しくない項目、存在しない color・yarn・pattern を参照している項目、count が 1 未満、cols が 1〜8 の外、rows が 1 未満の項目は読み飛ばし、`problems` に理由を1行ずつ残す。例外は出さない。
- `getContent().problems` が空でないときは、T1-10 で管理者メニューのログに warn で書く(このタスクでは不要)。

## テスト

1. 本物の4つの JSON を読み込むと、problems が空で、お題が5件、柄が5件ある。
2. 存在しない yarn を参照する柄は読み飛ばされ、problems に1行入る。
3. cols が 9 のお題は読み飛ばされる。
4. 同じ品番が2つの糸に使われていたら、後の方を読み飛ばして problems に入れる(品番で見分けるゲームのため、品番は重複させない)。
